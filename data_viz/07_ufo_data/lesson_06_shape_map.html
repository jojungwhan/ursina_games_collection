    Step 2: 모양별 색상 지도 🗺️

    각 모양을 다른 색상으로 표시한 지도를 만들어봅시다!

    **미션**: 색상으로 UFO 모양을 구분하세요 
    from leaflet_python import Map

# 모양별 색상 정의
shape_colors = {
    "light": "#FFD700",      # 금색
    "triangle": "#9C27B0",   # 보라색
    "disk": "#2196F3",       # 파란색
    "fireball": "#FF5722",   # 주황색
    "circle": "#4CAF50",     # 초록색
}

shape_emoji = {
    "light": "🔦", "triangle": "🔺", "disk": "🛸",
    "fireball": "🔥", "circle": "⭕"
}

# 다양한 모양의 UFO 목격 데이터
sightings = [
    {"shape": "light", "lat": 33.45, "lng": -112.07, "city": "Phoenix"},
    {"shape": "light", "lat": 36.17, "lng": -115.14, "city": "Las Vegas"},
    {"shape": "triangle", "lat": 40.71, "lng": -74.01, "city": "New York"},
    {"shape": "triangle", "lat": 41.88, "lng": -87.63, "city": "Chicago"},
    {"shape": "disk", "lat": 34.05, "lng": -118.24, "city": "Los Angeles"},
    {"shape": "disk", "lat": 47.61, "lng": -122.33, "city": "Seattle"},
    {"shape": "fireball", "lat": 29.76, "lng": -95.37, "city": "Houston"},
    {"shape": "fireball", "lat": 25.76, "lng": -80.19, "city": "Miami"},
    {"shape": "circle", "lat": 39.74, "lng": -104.99, "city": "Denver"},
    {"shape": "circle", "lat": 33.75, "lng": -84.39, "city": "Atlanta"},
]

m = Map(center=[39, -98], zoom=4)

for ufo in sightings:
    color = shape_colors.get(ufo['shape'], 'gray')
    emoji = shape_emoji.get(ufo['shape'], '❓')

    # 색상별 원 추가
    m.add_circle(
        center=[ufo['lat'], ufo['lng']],
        radius=80000,  # 80km
        color=color,
        fill=True,
        fill_opacity=0.6,
        popup=f"{emoji} {ufo['shape']} - {ufo['city']}"
    )

m.show()

print("색상별로 UFO 모양이 다르게 표시돼요!")
print("🔦 금색=Light, 🔺 보라=Triangle, 🛸 파랑=Disk")
