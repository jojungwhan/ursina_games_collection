    Feasibility Study: Python Turtle Auto-Grader System

    Date: 2026-01-18 Status: Feasibility Analysis (No Implementation) Target Location: After "나만의 우주 만들기" section

    Executive Summary

    This document analyzes the feasibility of adding an auto-grading system for Python turtle graphics quizzes. The study compares two architectural approaches: the server-side "headless" grader as originally proposed, and a client-side adaptation that integrates with the existing Skulpt-based infrastructure.

    Key Finding: The proposed server-side architecture is technically feasible but represents a significant departure from the current client-side-only architecture. A hybrid approach is recommended.

    1. Current Architecture Analysis

    1.1 Existing Technology Stack

    Component Technology Execution
    Python Runtime Skulpt (JavaScript) Browser
    Turtle Graphics Skulpt Turtle Module Browser Canvas
    Quiz System quiz-engine.js, cloze_quiz.js Browser
    Site Generator MkDocs + Material Static Build
    Server None (static hosting) N/A

    1.2 Key Constraint: No Server-Side Python

    The current site is 100% client-side: - MkDocs generates static HTML/CSS/JS - Skulpt executes Python in the browser - No backend server processes student code - Hosted on static file servers (GitHub Pages, Netlify, etc.)

    The gap: Adding server-side grading requires: 1. A Python backend server (Flask, FastAPI, etc.) 2. Server hosting infrastructure 3. Code submission API 4. Security sandboxing for student code 5. Session/authentication handling

    3.2 Architecture Options

    Option A: Full Server-Side (As Proposed)

    Pros: - Can run real Python (not limited by Skulpt) - Proper sandboxing with Docker/chroot - Can handle complex test cases - Industry-standard grading approach

    Cons: - Requires server infrastructure ($) - Adds latency (network round-trip) - Security concerns (running student code) - Maintenance overhead - Breaks current "static site" simplicity

    Estimated Effort: 3-4 weeks

    Option B: Client-Side Adaptation

    Adapt the MockTurtle concept to work within Skulpt/JavaScript:

    Cons: - Limited to Skulpt's Python subset - Students could inspect grading logic - Less secure - Can't test file I/O or imports

    Estimated Effort: 1-2 weeks

    4.1 Client-Side MockTurtle (JavaScript)

    // mock_turtle_analyzer.js

class MockTurtle {
    constructor() {
        this.x = 0;
        this.y = 0;
        this.heading = 90; // Facing up (standard turtle orientation)
        this.lines = [];
        this.penIsDown = true;
        this.startX = 0;
        this.startY = 0;
    }

    forward(distance) {
        const rad = (this.heading * Math.PI) / 180;
        const newX = this.x + distance * Math.cos(rad);
        const newY = this.y + distance * Math.sin(rad);

        if (this.penIsDown) {
            this.lines.push({
                x1: this.x, y1: this.y,
                x2: newX, y2: newY,
                length: distance
            });
        }

        this.x = newX;
        this.y = newY;
    }

    backward(distance) { this.forward(-distance); }

    right(angle) { this.heading = (this.heading - angle) % 360; }
    left(angle) { this.heading = (this.heading + angle) % 360; }

    penup() { this.penIsDown = false; }
    pendown() { this.penIsDown = true; }

    goto(x, y) {
        if (this.penIsDown) {
            const dx = x - this.x;
            const dy = y - this.y;
            this.lines.push({
                x1: this.x, y1: this.y,
                x2: x, y2: y,
                length: Math.sqrt(dx*dx + dy*dy)
            });
        }
        this.x = x;
        this.y = y;
    }

    // No-op methods (visual only, don't affect grading)
    speed(s) {}
    pencolor(c) {}
    pensize(w) {}
    shape(s) {}
    hideturtle() {}
    showturtle() {}
}

function analyzeShape(mockTurtle) {
    const lines = mockTurtle.lines;
    const EPSILON = 0.001; // Floating point tolerance

    // Calculate total perimeter
    const totalPerimeter = lines.reduce((sum, line) => sum + line.length, 0);

    // Calculate average side length
    const avgSideLength = lines.length > 0 ? totalPerimeter / lines.length : 0;

    // Check if closed (end position near start)
    const dx = mockTurtle.x - mockTurtle.startX;
    const dy = mockTurtle.y - mockTurtle.startY;
    const distanceFromStart = Math.sqrt(dx*dx + dy*dy);
    const isClosed = distanceFromStart < EPSILON;

    // Check if all sides are equal length (regular polygon)
    const sideLengths = lines.map(l => l.length);
    const isRegular = sideLengths.every(len =>
        Math.abs(len - avgSideLength) < EPSILON
    );

    return {
        side_count: lines.length,
        total_perimeter: Math.round(totalPerimeter * 100) / 100,
        avg_side_length: Math.round(avgSideLength * 100) / 100,
        is_closed_loop: isClosed,
        is_regular_polygon: isRegular,
        final_position: { x: mockTurtle.x, y: mockTurtle.y },
        lines: lines
    };
}

    4.2 Integration with Existing Quiz System

    // turtle_quiz_grader.js

class TurtleQuizGrader {
    constructor(criteria) {
        this.criteria = criteria;
        // Example criteria:
        // { shape: 'square', side_length: 100, tolerance: 1 }
        // { shape: 'hexagon', side_length: 50, must_be_closed: true }
    }

    grade(studentCode) {
        const mock = new MockTurtle();

        try {
            // Execute student's draw_shape function
            // This would need Skulpt integration
            this.executeInSkulpt(studentCode, mock);
        } catch (error) {
            return {
                passed: false,
                score: 0,
                feedback: `코드 실행 오류: ${error.message}`,
                error: true
            };
        }

        const analysis = analyzeShape(mock);
        return this.evaluateCriteria(analysis);
    }

    evaluateCriteria(analysis) {
        const results = [];
        let totalScore = 0;
        let maxScore = 0;

        // Check side count
        if (this.criteria.side_count !== undefined) {
            maxScore += 25;
            if (analysis.side_count === this.criteria.side_count) {
                totalScore += 25;
                results.push({ test: '변 개수', passed: true, msg: `✅ ${analysis.side_count}개` });
            } else {
                results.push({ test: '변 개수', passed: false,
                    msg: `❌ 예상: ${this.criteria.side_count}개, 실제: ${analysis.side_count}개` });
            }
        }

        // Check side length
        if (this.criteria.side_length !== undefined) {
            maxScore += 25;
            const tolerance = this.criteria.tolerance || 1;
            if (Math.abs(analysis.avg_side_length - this.criteria.side_length) <= tolerance) {
                totalScore += 25;
                results.push({ test: '변 길이', passed: true, msg: `✅ ${analysis.avg_side_length}` });
            } else {
                results.push({ test: '변 길이', passed: false,
                    msg: `❌ 예상: ${this.criteria.side_length}, 실제: ${analysis.avg_side_length}` });
            }
        }

        // Check if closed
        if (this.criteria.must_be_closed) {
            maxScore += 25;
            if (analysis.is_closed_loop) {
                totalScore += 25;
                results.push({ test: '도형 닫힘', passed: true, msg: '✅ 도형이 닫혀 있습니다' });
            } else {
                results.push({ test: '도형 닫힘', passed: false, msg: '❌ 도형이 닫히지 않았습니다' });
            }
        }

        // Check if regular
        if (this.criteria.must_be_regular) {
            maxScore += 25;
            if (analysis.is_regular_polygon) {
                totalScore += 25;
                results.push({ test: '정다각형', passed: true, msg: '✅ 모든 변의 길이가 같습니다' });
            } else {
                results.push({ test: '정다각형', passed: false, msg: '❌ 변의 길이가 일정하지 않습니다' });
            }
        }

        const percentage = maxScore > 0 ? Math.round((totalScore / maxScore) * 100) : 0;

        return {
            passed: totalScore === maxScore,
            score: percentage,
            totalScore: totalScore,
            maxScore: maxScore,
            results: results,
            analysis: analysis
        };
    }
}

import math
import sys
from unittest.mock import MagicMock

# Prevent actual turtle import from crashing
sys.modules['turtle'] = MagicMock()

class MockTurtle:
    """Headless turtle that tracks drawing without display."""

    def __init__(self):
        self.x = 0.0
        self.y = 0.0
        self.heading = 90.0  # Facing up
        self.pen_down = True
        self.lines = []  # List of ((x1,y1), (x2,y2)) tuples
        self.start_pos = (0.0, 0.0)

    def forward(self, distance):
        rad = math.radians(self.heading)
        new_x = self.x + distance * math.cos(rad)
        new_y = self.y + distance * math.sin(rad)

        if self.pen_down:
            self.lines.append(((self.x, self.y), (new_x, new_y)))

        self.x, self.y = new_x, new_y

    def backward(self, distance):
        self.forward(-distance)

    def right(self, angle):
        self.heading = (self.heading - angle) % 360

    def left(self, angle):
        self.heading = (self.heading + angle) % 360

    def goto(self, x, y):
        if self.pen_down:
            self.lines.append(((self.x, self.y), (x, y)))
        self.x, self.y = float(x), float(y)

    def setpos(self, x, y):
        self.goto(x, y)

    def setposition(self, x, y):
        self.goto(x, y)

    def penup(self):
        self.pen_down = False

    def pendown(self):
        self.pen_down = True

    def pu(self):
        self.penup()

    def pd(self):
        self.pendown()

def analyze_shape(mock_turtle):
    """Analyze the shape drawn by MockTurtle (rotation-agnostic)."""
    lines = mock_turtle.lines

    if not lines:
        return {
            'side_count': 0,
            'total_perimeter': 0.0,
            'avg_side_length': 0.0,
            'is_closed_loop': True,
            'is_regular_polygon': False
        }

    # Calculate line lengths
    def line_length(line):
        (x1, y1), (x2, y2) = line
        return math.sqrt((x2 - x1)**2 + (y2 - y1)**2)

    lengths = [line_length(line) for line in lines]
    total_perimeter = sum(lengths)
    avg_length = total_perimeter / len(lengths)

    # Check if closed (end near start)
    EPSILON = 0.001
    dx = mock_turtle.x - mock_turtle.start_pos[0]
    dy = mock_turtle.y - mock_turtle.start_pos[1]
    is_closed = math.sqrt(dx**2 + dy**2) < EPSILON

    # Check if regular (all sides equal)
    is_regular = all(abs(length - avg_length) < EPSILON for length in lengths)

    return {
        'side_count': len(lines),
        'total_perimeter': round(total_perimeter, 2),
        'avg_side_length': round(avg_length, 2),
        'is_closed_loop': is_closed,
        'is_regular_polygon': is_regular
    }

import sys
import importlib.util
from mock_turtle import MockTurtle, analyze_shape

def grade_submission(student_file_path, criteria):
    """
    Grade a student submission.

    Args:
        student_file_path: Path to student's Python file
        criteria: Dict with grading criteria
            Example: {'shape': 'square', 'side_count': 4, 'side_length': 100}

    Returns:
        Dict with grading results
    """
    # Create mock turtle
    mock = MockTurtle()

    # Load and run student code
    try:
        spec = importlib.util.spec_from_file_location("student_code", student_file_path)
        student_module = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(student_module)

        # Check if draw_shape exists
        if not hasattr(student_module, 'draw_shape'):
            return {
                'passed': False,
                'score': 0,
                'error': 'draw_shape 함수를 찾을 수 없습니다'
            }

        # Execute student's function
        student_module.draw_shape(mock)

    except SyntaxError as e:
        return {
            'passed': False,
            'score': 0,
            'error': f'문법 오류 (줄 {e.lineno}): {e.msg}'
        }
    except Exception as e:
        return {
            'passed': False,
            'score': 0,
            'error': f'실행 오류: {str(e)}'
        }

    # Analyze the shape
    analysis = analyze_shape(mock)

    # Grade against criteria
    results = []
    total_points = 0
    max_points = 0

    if 'side_count' in criteria:
        max_points += 25
        if analysis['side_count'] == criteria['side_count']:
            total_points += 25
            results.append(f"✅ 변 개수: {analysis['side_count']}개 (정답)")
        else:
            results.append(f"❌ 변 개수: 예상 {criteria['side_count']}개, 실제 {analysis['side_count']}개")

    if 'side_length' in criteria:
        max_points += 25
        tolerance = criteria.get('tolerance', 1)
        if abs(analysis['avg_side_length'] - criteria['side_length']) <= tolerance:
            total_points += 25
            results.append(f"✅ 변 길이: {analysis['avg_side_length']} (정답)")
        else:
            results.append(f"❌ 변 길이: 예상 {criteria['side_length']}, 실제 {analysis['avg_side_length']}")

    if criteria.get('must_be_closed', False):
        max_points += 25
        if analysis['is_closed_loop']:
            total_points += 25
            results.append("✅ 도형이 닫혀 있습니다")
        else:
            results.append("❌ 도형이 닫히지 않았습니다")

    if criteria.get('must_be_regular', False):
        max_points += 25
        if analysis['is_regular_polygon']:
            total_points += 25
            results.append("✅ 정다각형입니다")
        else:
            results.append("❌ 변의 길이가 일정하지 않습니다")

    score = (total_points / max_points * 100) if max_points > 0 else 0

    return {
        'passed': total_points == max_points,
        'score': round(score),
        'results': results,
        'analysis': analysis
    }
